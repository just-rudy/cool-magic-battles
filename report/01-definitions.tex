\begin{definitions}
	\definition{Игрок}{участник игровой сессии, обладающий параметрами состояния (здоровье, размер руки, эхо, очки крутости) и взаимодействующий с игровыми объектами в процессе игры}
	
	\definition{Очки здоровья}{характеристика игрока, определяющая количество урона, которое он может получить до смерти}
	
	\definition{Очки эхо}{игровой ресурс, накапливаемый в ходе игры и используемый для покупки карт}
	
	\definition{Памятка о бренности бытия}{игровой элемент, фиксирующий факт смерти игрока и влияющий на итоговый результат партии}
	
	\definition{Очки крутости}{показатель, определяющий итоговый результат игрока в конце игры}
	
	\definition{Карта}{игровой объект, обладающий набором характеристик и эффектом, применяемым при розыгрыше}
	
	\definition{Эффект карты}{действие, выполняемое при использовании карты (например, атака, защита, добор карт)}
	
	\definition{Стоимость карты}{количество очков эхо, необходимое для приобретения карты}
\end{definitions}
